lights = [
[1,1,1],
[1,1,1],
[1,1,1]  
]
switches = []
for i in range(3):
  row = list(map(int,input().split()))
  switches.append(row)
for i in range(3):
  for j in range(3):
    if (switches[i][j]%2 != 0) : 
      lights[i][j] = int(not(lights[i][j]))
      if i>0:
        lights[i-1][j] = int(not(lights[i-1][j]))
      if i <2:
        lights[i+1][j] = int(not(lights[i+1][j]))
      if j > 0 :
        lights[i][j-1] = int(not(lights[i][j-1]))
      if j < 2:
        lights[i][j+1] = int(not(lights[i][j+1]))
 
for i in range(3):
  print(*lights[i],sep="")
