N = int(input())
w = list(map(int,input().split()))
w.sort()
M = 2*N
ans = 10**18
 
for i in range(M):
  for j in range(i+1, M):
    lst = []
    for k in range(M):
      if k != i and k != j:
        lst.append(w[k])
    current = 0
    for k in range(0,len(lst),2):
      current += abs(lst[k]-lst[k+1])
    ans = min(ans,current)
 
print(ans)
