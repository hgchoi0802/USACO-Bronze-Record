n = int(input())
a = list(map(int,input().split()))
cnt = 0
for i in range(n):
  if a[i] == 1:
    cnt += 1
ans = 0
for i in range(n):
  for j in range(n):
    temp = cnt
    if i > j:
      continue
    for k in range(i,j+1):
      if a[k] == 1:
        temp -= 1
      else:
        temp += 1
    ans = max(ans,temp)
print(ans)
