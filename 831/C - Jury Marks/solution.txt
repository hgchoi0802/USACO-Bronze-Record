k, n = map(int, input().split())
 
a = list(map(int, input().split()))
b = list(map(int, input().split()))
 
for i in range(1,k) : 
  a[i] += a[i-1]
  
ans = set()
 
for i in range(n):
  tmp = set() 
  for j in range(k):
    if i == 0 : 
      tmp.add(b[i]-a[j])
      continue
    if b[i]-a[j] in ans : 
      tmp.add(b[i]-a[j])
  ans = tmp
print(len(ans))
