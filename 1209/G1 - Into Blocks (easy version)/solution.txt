n, q = map(int, input().split())
a = list(map(int, input().split()))
 
first = {}
last = {}
 
for i in range(n):
    if a[i] not in first:
        first[a[i]] = i
    last[a[i]] = i
 
blocks = []
 
for x in first:
    blocks.append([first[x], last[x]])
 
blocks.sort()
 
merged = []
 
for i in range(len(blocks)):
    L = blocks[i][0]
    R = blocks[i][1]
 
    if len(merged) == 0:
        merged.append([L, R])
    elif L <= merged[-1][1]:
        merged[-1][1] = max(merged[-1][1], R)
    else:
        merged.append([L, R])
 
ans = 0
 
for i in range(len(merged)):
    L = merged[i][0]
    R = merged[i][1]
 
    count = {}
 
    for j in range(L, R + 1):
        if a[j] not in count:
            count[a[j]] = 0
        count[a[j]] += 1
 
    best = 0
 
    for x in count:
        best = max(best, count[x])
 
    ans += R - L + 1 - best
 
print(ans)
