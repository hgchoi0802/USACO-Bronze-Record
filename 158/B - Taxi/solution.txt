s = int(input())
lst = list(map(int,input().split()))
groupcnt = [0]*4
for i in range(s):
  groupcnt[lst[i]-1] += 1
t = groupcnt[2] + groupcnt[3]
groupcnt[0] = max(0,groupcnt[0] - groupcnt[2]) 
t += groupcnt[1] // 2 
groupcnt[1] %= 2 
 
if groupcnt[1] : 
  t += 1 
  groupcnt[0] = max(0,groupcnt[0] - 2)
 
if groupcnt[0] > 0 : 
  t += (groupcnt[0] + 3) // 4 
 
print(t)
