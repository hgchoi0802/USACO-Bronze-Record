t = int(input())
 
for i in range(t):
    n = int(input())
    a = [0] + list(map(int, input().split()))
 
    pos = [0] * (2 * n + 1)
 
    for j in range(1, n + 1):
        pos[a[j]] = j
 
    ans = 0
 
    for j in range(1, n + 1):
        x = a[j]
 
        for product in range(x, 2 * n + 1, x):
            y = product // x
 
            if y <= 2 * n and pos[y] != 0:
                k = pos[y]
 
                if j < k and j + k == product:
                    ans += 1
 
    print(ans)
